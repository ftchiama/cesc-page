```sql
-- 1. Listar Alocações
CREATE OR REPLACE FUNCTION cesc.cesc_diretor_listar_alocacoes(p_token varchar, p_ano_lectivo varchar)
RETURNS json AS $$
DECLARE v_user cesc.utilizadores; v_lista json;
BEGIN
    v_user := cesc.validar_sessao_role(p_token, ARRAY['diretor','sub_diretor']);
    IF v_user IS NULL THEN RETURN json_build_object('ok', false, 'erro', 'Não autorizado.'); END IF;

    SELECT json_agg(t) INTO v_lista FROM (
        SELECT * FROM cesc.alocacao_salas 
        WHERE ano_lectivo = p_ano_lectivo 
        ORDER BY turno, pavilhao, sala_inicio
    ) t;

    RETURN json_build_object('ok', true, 'alocacoes', COALESCE(v_lista, '[]'::json));
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE OR REPLACE FUNCTION public.cesc_diretor_listar_alocacoes(p_token varchar, p_ano_lectivo varchar)
RETURNS json AS $$ BEGIN RETURN cesc.cesc_diretor_listar_alocacoes(p_token, p_ano_lectivo); END; $$ LANGUAGE plpgsql SECURITY DEFINER;
GRANT EXECUTE ON FUNCTION public.cesc_diretor_listar_alocacoes TO anon, authenticated;


-- 2. Guardar Alocação
CREATE OR REPLACE FUNCTION cesc.cesc_diretor_guardar_alocacao(
    p_token varchar, p_ano_lectivo varchar, p_turno varchar, p_pavilhao varchar, 
    p_sala_inicio int, p_sala_fim int, p_classe varchar, p_curso varchar
) RETURNS json AS $$
DECLARE v_user cesc.utilizadores;
BEGIN
    v_user := cesc.validar_sessao_role(p_token, ARRAY['diretor','sub_diretor']);
    IF v_user IS NULL THEN RETURN json_build_object('ok', false, 'erro', 'Não autorizado.'); END IF;

    INSERT INTO cesc.alocacao_salas (ano_lectivo, turno, pavilhao, sala_inicio, sala_fim, classe, curso)
    VALUES (p_ano_lectivo, p_turno, p_pavilhao, p_sala_inicio, p_sala_fim, p_classe, p_curso)
    ON CONFLICT (ano_lectivo, turno, pavilhao, sala_inicio, sala_fim, classe, curso) DO NOTHING;

    RETURN json_build_object('ok', true);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE OR REPLACE FUNCTION public.cesc_diretor_guardar_alocacao(varchar, varchar, varchar, varchar, int, int, varchar, varchar)
RETURNS json AS $$ BEGIN RETURN cesc.cesc_diretor_guardar_alocacao($1,$2,$3,$4,$5,$6,$7,$8); END; $$ LANGUAGE plpgsql SECURITY DEFINER;
GRANT EXECUTE ON FUNCTION public.cesc_diretor_guardar_alocacao TO anon, authenticated;
```

```javascript
let regrasAuraCache = [];
let alocacoesAuraCache = [];

async function carregarAuraDir() {
    const token = obterToken();
    const containerRegras = document.getElementById("lista-regras-aura");
    containerRegras.innerHTML = '<div class="loader-container" style="min-height:100px;"><div class="loader"></div></div>';

    // 1. Carregar Regras
    const resRegras = await chamarRPC("cesc_diretor_listar_regras_aura", { p_token: token, p_ano_lectivo: "2026/2027" });
    if (resRegras.ok) {
        regrasAuraCache = resRegras.regras || [];
        renderizarRegrasAura(regrasAuraCache);
    } else {
        containerRegras.innerHTML = '<div class="estado-vazio">' + esc(resRegras.erro) + '</div>';
    }

    // 2. Carregar Alocações
    const selAno = document.getElementById("filtro-aura-ano-alocacao");
    if (selAno && !selAno.value) {
        selAno.innerHTML = '<option value="2026/2027">2026/2027</option>';
    }
    const ano = selAno ? selAno.value : "2026/2027";
    
    const containerAloc = document.getElementById("lista-alocacoes-aura");
    containerAloc.innerHTML = '<div class="loader-container" style="min-height:100px;"><div class="loader"></div></div>';

    const resAloc = await chamarRPC("cesc_diretor_listar_alocacoes", { p_token: token, p_ano_lectivo: ano });
    if (resAloc.ok) {
        alocacoesAuraCache = resAloc.alocacoes || [];
        renderizarAlocacoesAura(alocacoesAuraCache);
    } else {
        containerAloc.innerHTML = '<div class="estado-vazio">' + esc(resAloc.erro) + '</div>';
    }

    // Configurar eventos (só uma vez)
    if (!window.auraLigado) {
        window.auraLigado = true;
        
        const selAnoAloc = document.getElementById("filtro-aura-ano-alocacao");
        if (selAnoAloc) selAnoAloc.addEventListener("change", carregarAuraDir);

        document.getElementById("btn-executar-aura").addEventListener("click", async function () {
            if (!confirm("Tens a certeza que queres executar a AURA? Isto vai distribuir os alunos e mudar os estados das inscrições.")) return;

            const btn = this;
            const msg = document.getElementById("msg-executar-aura");
            btn.disabled = true;
            btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> A processar...';
            msg.textContent = "";

            const res = await chamarRPC("cesc_executar_aura", { p_token: obterToken(), p_ano_lectivo: "2026/2027" });

            btn.disabled = false;
            btn.innerHTML = '<i class="fas fa-magic"></i> Executar AURA';

            if (res.ok) {
                msg.className = "msg-form sucesso";
                msg.textContent = "AURA executada! Alunos processados: " + res.total_processados + ". Aprovados: " + res.aprovados + ". Reprovados: " + res.reprovados + ".";
            } else {
                msg.className = "msg-form erro";
                msg.textContent = res.erro || "Erro ao executar AURA.";
            }
        });
    }
}

}

async function guardarRegraAura(idx) {
    const token = obterToken();
    const regra = regrasAuraCache[idx];
    const crit = parseInt(document.getElementById("regra-crit-" + idx).value, 10);
    const sec = parseInt(document.getElementById("regra-sec-" + idx).value, 10);

    const res = await chamarRPC("cesc_diretor_atualizar_regra_aura", {
        p_token: token, p_regra_id: regra.id, p_max_crit: crit, p_max_sec: sec
    });
    if (res.ok) alert("Regra guardada com sucesso.");
    else alert(res.erro);
}
```

1. SQL: Criar as RPCs de Alocação (para resolver o 404)

Cola isto no SQL Editor do Supabase. Estas funções vão permitir listar e guardar as alocações que o Diretor definir.

```sql
-- 1. Listar Alocações
DROP FUNCTION IF EXISTS cesc.cesc_diretor_listar_alocacoes(varchar, varchar) CASCADE;
DROP FUNCTION IF EXISTS public.cesc_diretor_listar_alocacoes(varchar, varchar) CASCADE;

CREATE OR REPLACE FUNCTION cesc.cesc_diretor_listar_alocacoes(p_token varchar, p_ano_lectivo varchar)
RETURNS json AS $$
DECLARE v_user cesc.utilizadores; v_lista json;
BEGIN
    v_user := cesc.validar_sessao_role(p_token, ARRAY['diretor','sub_diretor']);
    IF v_user IS NULL THEN RETURN json_build_object('ok', false, 'erro', 'Não autorizado.'); END IF;

    SELECT json_agg(t) INTO v_lista FROM (
        SELECT * FROM cesc.alocacao_salas 
        WHERE ano_lectivo = p_ano_lectivo 
        ORDER BY turno, pavilhao, sala_inicio
    ) t;

    RETURN json_build_object('ok', true, 'alocacoes', COALESCE(v_lista, '[]'::json));
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE OR REPLACE FUNCTION public.cesc_diretor_listar_alocacoes(p_token varchar, p_ano_lectivo varchar)
RETURNS json AS $$ BEGIN RETURN cesc.cesc_diretor_listar_alocacoes(p_token, p_ano_lectivo); END; $$ LANGUAGE plpgsql SECURITY DEFINER;
GRANT EXECUTE ON FUNCTION public.cesc_diretor_listar_alocacoes TO anon, authenticated;

-- 2. Guardar Alocação
DROP FUNCTION IF EXISTS cesc.cesc_diretor_guardar_alocacao(varchar, varchar, varchar, varchar, int, int, varchar, varchar) CASCADE;
DROP FUNCTION IF EXISTS public.cesc_diretor_guardar_alocacao(varchar, varchar, varchar, varchar, int, int, varchar, varchar) CASCADE;

CREATE OR REPLACE FUNCTION cesc.cesc_diretor_guardar_alocacao(
    p_token varchar, p_ano_lectivo varchar, p_turno varchar, p_pavilhao varchar, 
    p_sala_inicio int, p_sala_fim int, p_classe varchar, p_curso varchar
) RETURNS json AS $$
DECLARE v_user cesc.utilizadores;
BEGIN
    v_user := cesc.validar_sessao_role(p_token, ARRAY['diretor','sub_diretor']);
    IF v_user IS NULL THEN RETURN json_build_object('ok', false, 'erro', 'Não autorizado.'); END IF;

    INSERT INTO cesc.alocacao_salas (ano_lectivo, turno, pavilhao, sala_inicio, sala_fim, classe, curso)
    VALUES (p_ano_lectivo, p_turno, p_pavilhao, p_sala_inicio, p_sala_fim, p_classe, p_curso)
    ON CONFLICT (ano_lectivo, turno, pavilhao, sala_inicio, sala_fim, classe, curso) DO NOTHING;

    RETURN json_build_object('ok', true);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE OR REPLACE FUNCTION public.cesc_diretor_guardar_alocacao(varchar, varchar, varchar, varchar, int, int, varchar, varchar)
RETURNS json AS $$ BEGIN RETURN cesc.cesc_diretor_guardar_alocacao($1,$2,$3,$4,$5,$6,$7,$8); END; $$ LANGUAGE plpgsql SECURITY DEFINER;
GRANT EXECUTE ON FUNCTION public.cesc_diretor_guardar_alocacao TO anon, authenticated;
```

```javascript
async function carregarAuraDir() {

    }

    // 2. CARREGAR ALOCAÇÕES
    const selAno = document.getElementById("filtro-aura-ano-alocacao");

    }

    // 3. EVENTOS
    if (!window.auraLigado) {
        window.auraLigado = true;
        
        const selAnoAloc = document.getElementById("filtro-aura-ano-alocacao");
        if (selAnoAloc) selAnoAloc.addEventListener("change", carregarAuraDir);

                const btn = this;
                const msg = document.getElementById("msg-executar-aura");
                btn.disabled = true;
                btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> A processar...';
                msg.textContent = "";

                const res = await chamarRPC("cesc_executar_aura", { p_token: obterToken(), p_ano_lectivo: "2026/2027" });

                } else {
                    msg.className = "msg-form erro";
                    msg.textContent = res.erro || "Erro ao executar AURA.";
                }
            });
        }
```

```javascript
async function executarAura(rpcName, titulo) {
    const btn = document.getElementById("btn-executar-" + (rpcName === "cesc_aura_julgar" ? "juiz" : "distribuidor"));
    const msg = document.getElementById("msg-executar-aura");
    const txtOriginal = btn.innerHTML;
    
    btn.disabled = true;
    btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> A processar...';
    msg.textContent = "";
    msg.className = "msg-form";

    const res = await chamarRPC(rpcName, { p_token: obterToken(), p_ano_lectivo: "2026/2027" });

    btn.disabled = false;
    btn.innerHTML = txtOriginal;

    if (res.ok) {
        msg.className = "msg-form sucesso";

            msg.textContent = `${titulo} concluído! Total: ${res.total}, Aprovados: ${res.aprovados}, Reprovados: ${res.reprovados}.`;
```

```javascript
// Função simples para criar um modal de confirmação sem depender do core
function abrirModalConfirmacao(titulo, mensagem, acao) {
    const overlay = document.getElementById("modal-overlay");
    if (!overlay) return;

    document.getElementById("modal-titulo").textContent = titulo;
    document.getElementById("modal-corpo").innerHTML = '<p>' + mensagem + '</p>';
    
    const btnConfirmar = document.getElementById("modal-confirmar");
    const btnCancelar = document.getElementById("modal-cancelar");
    
    btnConfirmar.style.display = "block";
    btnConfirmar.textContent = "Confirmar";
    
    // Clonar para remover eventos antigos
    const novoBtn = btnConfirmar.cloneNode(true);
    btnConfirmar.parentNode.replaceChild(novoBtn, btnConfirmar);
    
    novoBtn.addEventListener("click", async function () {
        novoBtn.disabled = true;
        novoBtn.textContent = "A processar...";
        await acao();
        novoBtn.disabled = false;
        novoBtn.textContent = "Confirmar";
        overlay.style.display = "none";
    });

    btnCancelar.onclick = function() { overlay.style.display = "none"; };
    document.getElementById("modal-fechar").onclick = function() { overlay.style.display = "none"; };

    overlay.style.display = "flex";
}

async function executarAura(rpcName, titulo) {

    btn.disabled = false;
    btn.innerHTML = txtOriginal;
```

Aqui está a diferença exata, segundo o AURAv1.md e o modelo de dados:

1. estado (Estado da Inscrição)

Refere-se à fase do processo de candidatura.

· Onde o candidato está no funil (a_pagar, pendente, aprovada, matricula_paga, matriculado, rejeitada).
· Quem controla: O sistema (pagamento) e a Secretária (aprovação/rejeição).
· Exemplo: O aluno pagou a matrícula, logo o estado é matricula_paga.

2. tipo (Tipo de Aluno para o AURA)

Refere-se à origem/classificação do aluno para o algoritmo de distribuição.

· O AURA precisa de saber se o aluno é novo, se já estudava na escola e passou de ano, ou se reprovou.
· Valores possíveis (definidos no AURAv1.md): novo, aprovado, reprovado.
· Quem controla: O JUIZ (que lê as notas e decide se é aprovado ou reprovado) e a Secretária (quando cria a inscrição de um aluno novo, define tipo = 'novo').

Como se relacionam no fluxo:

1. O aluno inscreve-se. O estado é a_pagar e o tipo é novo (por defeito).
2. O aluno paga a inscrição e a matrícula. O estado passa a matricula_paga. O tipo continua novo.
3. O Diretor corre o JUIZ. O JUIZ lê as notas e altera o tipo para aprovado ou reprovado. O estado não muda (continua matricula_paga).
4. O Diretor corre o DISTRIBUIDOR. Ele procura alunos com estado = 'matricula_paga' e coloca-os nas salas. (O tipo não influencia a distribuição em si, apenas diz se o aluno passou ou não para efeitos de transição).
5. Após a distribuição, a Secretária (ou o AURA) muda o estado para matriculado.

Resumindo: O estado diz "em que fase do processo a candidatura está". O tipo diz "qual foi o veredito do JUIZ sobre a passagem de ano do aluno". São campos independentes que contam histórias diferentes.

```java
    // Lista de disciplinas consideradas críticas (conforme AURAv1.md)
    private static final List<String> DISCIPLINAS_CRITICAS = List.of("Matemática", "Biologia", "Química", "Física");

    public void avaliar(List<Inscricao> inscricoes, List<NotaTrimestral> notas, List<RegraAura> regras) {
        for (Inscricao inscricao : inscricoes) {
            // O Juiz só avalia alunos com matrícula paga
            if (!"matricula_paga".equals(inscricao.estado)) continue;

            // Encontra a regra para a classe/curso
            RegraAura regra = regras.stream()
                .filter(r -> r.classe.equals(inscricao.classePretendida) &&
                            (r.curso == null || r.curso.equals(inscricao.cursoPretendido)))
                .findFirst().orElse(null);

            if (regra == null) {
                // Sem regra, reprova por defeito
                inscricao.tipo = "reprovado";
                inscricao.motivoRejeicao = "Sem regra AURA configurada.";
                continue;
            }

            // Filtra as notas do aluno para o ano letivo
            List<NotaTrimestral> notasAluno = notas.stream()
                .filter(n -> n.alunoId.equals(inscricao.alunoId) && n.anoLectivo.equals(inscricao.anoLectivo))
                .toList();

            // Conta as negativas
            long negCriticas = notasAluno.stream()
                .filter(n -> n.mfd < 10.0 && DISCIPLINAS_CRITICAS.contains(n.disciplina))
                .count();

            long negSecundarias = notasAluno.stream()
                .filter(n -> n.mfd < 10.0 && !DISCIPLINAS_CRITICAS.contains(n.disciplina))
                .count();

            // Verifica se passou
            boolean aprovado = (negCriticas <= regra.maxNegativasCriticas) && 
                               (negSecundarias <= regra.maxNegativasSecundarias);

            inscricao.tipo = aprovado ? "aprovado" : "reprovado";
            if (!aprovado) inscricao.motivoRejeicao = "Reprovado pelas negativas (AURA).";
        }
    }
}
```

service/DistribuidorAura.java (A Distribuição)

```java
package service;

import model.*;
import java.util.*;
import java.util.stream.Collectors;

public class DistribuidorAura {

    public void distribuir(String anoLectivo, List<Inscricao> inscricoes, List<Matricula> matriculas, 
                           List<AlocacaoSala> alocacoes, List<Sala> salas) {
        
        int numeroSequencial = 1;
        int totalDistribuidos = 0;

        // Percorre cada alocação (ex: Pav 1, Sala 1, 10ª CFB)
        for (AlocacaoSala aloc : alocacoes) {
            if (!aloc.anoLectivo.equals(anoLectivo)) continue;

            // Encontra as salas físicas que pertencem a esta alocação
            List<Sala> salasAlocadas = salas.stream()
                .filter(s -> s.turno.equals(aloc.turno) && 
                             s.pavilhao.equals(aloc.pavilhao) &&
                             extrairNumeroSala(s.nomeSala) >= aloc.salaInicio &&
                             extrairNumeroSala(s.nomeSala) <= aloc.salaFim)
                .sorted(Comparator.comparing(Sala::getNomeSala))
                .toList();

            // Encontra os alunos que precisam de ser distribuídos nesta alocação
            List<Inscricao> alunosParaDistribuir = inscricoes.stream()
                .filter(i -> i.anoLectivo.equals(anoLectivo) &&
                             "matricula_paga".equals(i.estado) &&
                             i.classePretendida.equals(aloc.classe) &&
                             (aloc.curso == null || aloc.curso.equals(i.cursoPretendido)))
                .sorted(Comparator.comparing(i -> i.alunoId.toString()))
                .collect(Collectors.toList());

            // Remove alunos que já têm sala
            alunosParaDistribuir.removeIf(insc -> {
                Optional<Matricula> mat = matriculas.stream()
                    .filter(m -> m.alunoId.equals(insc.alunoId) && m.anoLectivo.equals(anoLectivo))
                    .findFirst();
                return mat.isPresent() && mat.get().salaId != null;
            });

            // Preenche as salas até à capacidade
            int idxAluno = 0;
            for (Sala sala : salasAlocadas) {
                int alunosNaSala = 0;
                
                while (idxAluno < alunosParaDistribuir.size() && alunosNaSala < sala.capacidade) {
                    Inscricao aluno = alunosParaDistribuir.get(idxAluno);
                    
                    // Atualiza a matrícula deste aluno
                    for (Matricula m : matriculas) {
                        if (m.alunoId.equals(aluno.alunoId) && m.anoLectivo.equals(anoLectivo)) {
                            m.salaId = sala.id;
                            m.numeroAluno = numeroSequencial;
                            break;
                        }
                    }
                    
                    idxAluno++;
                    alunosNaSala++;
                    numeroSequencial++;
                    totalDistribuidos++;
                }
                
                if (idxAluno >= alunosParaDistribuir.size()) break; // Já não há mais alunos
            }
        }
        
        System.out.println("AURA DISTRIBUIDOR: " + totalDistribuidos + " alunos colocados em salas.");
    }

    private int extrairNumeroSala(String nomeSala) {
        try {
            return Integer.parseInt(nomeSala.replaceAll("\\D", ""));
        } catch (NumberFormatException e) {
            return -1;
        }
    }
}
```

service/AuraEngine.java

```java
package service;

import model.*;
import java.util.List;

public class AuraEngine {
    
    private final JuizAura juiz = new JuizAura();
    private final DistribuidorAura distribuidor = new DistribuidorAura();

    // Método 1: Só o JUIZ (Fim do ano)
    public void executarJuiz(String anoLectivo, List<Inscricao> inscricoes, List<NotaTrimestral> notas, List<RegraAura> regras) {
        System.out.println("--- A INICIAR O JUIZ ---");
        juiz.avaliar(inscricoes, notas, regras);
        
        long aprovados = inscricoes.stream().filter(i -> "aprovado".equals(i.tipo)).count();
        long reprovados = inscricoes.stream().filter(i -> "reprovado".equals(i.tipo)).count();
        System.out.println("JUIZ concluído. Aprovados: " + aprovados + ", Reprovados: " + reprovados);
    }

    // Método 2: Só o DISTRIBUIDOR (Início do ano)
    public void executarDistribuidor(String anoLectivo, List<Inscricao> inscricoes, List<Matricula> matriculas, 
                                     List<AlocacaoSala> alocacoes, List<Sala> salas) {
        System.out.println("--- A INICIAR O DISTRIBUIDOR ---");
        distribuidor.distribuir(anoLectivo, inscricoes, matriculas, alocacoes, salas);
    }

    // Método 3: Executar TUDO (Cuidado: Apenas se as matrículas do novo ano já existirem)
    public void executarTudo(String anoLectivo, List<Inscricao> inscricoes, List<Matricula> matriculas,
                             List<NotaTrimestral> notas, List<RegraAura> regras, 
                             List<AlocacaoSala> alocacoes, List<Sala> salas) {
        executarJuiz(anoLectivo, inscricoes, notas, regras);
        executarDistribuidor(anoLectivo, inscricoes, matriculas, alocacoes, salas);
    }
}
```

1. O que a conversão para Java prova

O código Java que te dei é uma prova de conceito de como o algoritmo funciona. Ele mostra que a lógica do JUIZ e do DISTRIBUIDOR é perfeitamente clara:

· O JUIZ lê notas e aplica regras.
· O DISTRIBUIDOR lê alocações e preenche salas.

O Java é excelente para isto porque tem tipos fortes e é fácil de testar (como fiz no Main.java).

2. Porque é que NÃO deves substituir as tabelas por Java no CESC

O teu projeto já está construído sobre uma arquitetura sólida:

· Supabase (PostgreSQL) guarda o estado (alunos, notas, matrículas, salas).
· RPCs (PL/pgSQL) fazem os cálculos dentro da base de dados.
· Frontend (JS) chama as RPCs e mostra os resultados.

Se substituíres as tabelas por um serviço Java, vais ter de:

1. Duplicar a base de dados: O Java teria de ler os dados do Supabase para a memória, processar, e depois voltar a gravar (ou manter tudo em memória, o que é perigoso).
2. Criar um servidor Java: O Render está configurado para o tclabs-email-api (Node/Deno). Terias de criar outro serviço (ex: Spring Boot) e pagar por ele.
3. Perder a atomicidade: O Supabase faz tudo numa transação. O Java teria de gerir transações de rede (o que é complexo e propício a erros).
4. Perder o RLS (Row Level Security): Toda a segurança que construímos (tokens, validação de roles) teria de ser reescrita em Java.

3. O que o código Java REALMENTE mostra

O Java que te dei não substitui as tabelas. Ele apenas te mostra a lógica que as RPCs do teu Supabase já implementam. Repara que:

· O JuizAura em Java faz exatamente o mesmo que o cesc_aura_julgar em PL/pgSQL.
· O DistribuidorAura em Java faz exatamente o mesmo que o cesc_aura_distribuir em PL/pgSQL.

A única diferença é que o PL/pgSQL corre dentro da base de dados (é atómico, seguro e rápido), enquanto o Java corre fora (teria de fazer chamadas de rede).

4. Então para que serve o Java?

O Java serve para duas coisas:

1. Testar a lógica: Podes usar o Main.java para simular cenários (ex: "e se houver 40 alunos e só 30 lugares?") sem mexer na base de dados real. É um ambiente de testes.
2. Migrar no futuro: Se um dia o CESC crescer e precisar de um backend próprio (ex: uma API Java para uma app móvel), a lógica já está pronta.

5. A minha recomendação

Mantém o Supabase (RPCs) como está. O teu backend já tem:

· O JUIZ (cesc_aura_julgar) a funcionar.
· O DISTRIBUIDOR (cesc_aura_distribuir) a funcionar.

O Java que te dei pode ficar guardado na pasta docs/java-reference/ do projeto, como material de estudo. Mas o sistema em produção deve continuar a usar o Supabase.
Sim. O princípio é universal: o Java foi só o exemplo que usámos. A mesma regra aplica-se a Python, C#, Node.js, Go, PHP, Ruby ou qualquer outra linguagem de backend.

Aqui está a regra de ouro para o teu projeto:

1. Onde a linguagem deve correr (e onde não deve)

A linguagem (seja Java, Python, Node, etc.) não substitui a base de dados. Ela serve para orquestrar e comunicar.

· A Base de Dados (PostgreSQL/Supabase): É a fonte da verdade. Ela guarda os dados e faz as operações pesadas de forma atómica. No teu caso, as RPCs (cesc_aura_julgar, cesc_aura_distribuir) já fazem o trabalho pesado (o loop de alunos, a verificação de regras, a distribuição nas salas).
· A Linguagem de Backend (Java, Python, Node, etc.): Serve como uma ponte (API). Ela recebe o pedido do frontend, valida o token, e chama a RPC no Supabase. O trabalho pesado continua a ser feito pelo SQL.

2. Porque é que centralizar na Base de Dados é o mais seguro para o CESC?

1. Atomicidade (Tudo ou Nada): Se o servidor Java/Python for abaixo a meio da distribuição, podes ter metade dos alunos colocados nas salas e a outra metade não. No Supabase, se a RPC falhar, ele desfaz tudo (rollback) e a base de dados fica intacta.
2. Segurança (RLS): O Supabase tem o RLS (Row Level Security) que bloqueia acessos diretos. Se usares um backend Java, esse backend teria de ter a chave service_role (a chave mestre) para aceder aos dados. Se essa chave for exposta, o sistema inteiro fica comprometido. As RPCs atuais não precisam da chave mestre, porque são security definer e validam o token do utilizador.
3. Performance: Fazer 1000 inserções na base de dados via código Java (uma a uma) é muito mais lento do que fazer um UPDATE em massa dentro da RPC do SQL. O SQL é otimizado para isso.

3. Onde é que a linguagem (Java, Python, Node) é útil no CESC?

Não é para substituir as tabelas, mas para funções específicas que o SQL não é bom a fazer:

· Envio de E-mails: Já usas o Node/Deno (Edge Functions) e o Render (Node) para falar com o Brevo. Isso é perfeito. O SQL não envia e-mails.
· Integração com Pagamentos (AppyPay): O SQL não faz chamadas HTTP. Por isso, usas Edge Functions para falar com o AppyPay.
· Geração de PDFs: Se quiseres gerar certificados em PDF, o Python (com bibliotecas como ReportLab) ou o Node (com Puppeteer) são melhores do que o SQL.

4. O caso do Java (e de outras linguagens)

Se amanhã quiseres criar uma App Móvel (Android/iOS) para o CESC, vais precisar de um backend Java/Python/Node para servir a app. Nesse caso:

· O backend Java não vai substituir o Supabase.
· O backend Java vai apenas chamar as RPCs do Supabase e devolver os dados formatados para a app.
· A lógica de negócio (JUIZ e DISTRIBUIDOR) continua no SQL, porque é lá que ela está segura e testada.

A linguagem serve para a interface (o que o utilizador vê) e para integrações externas (e-mails, pagamentos, PDFs). A base de dados é o "cérebro" que guarda e processa a informação de forma segura.
